-- 1. Тип Топлива
CREATE TABLE FuelType (
    FuelType_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(100) NOT NULL
);

-- 2. Топливо
CREATE TABLE Fuel (
    Fuel_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(100) NOT NULL,
    FuelType_Id INT REFERENCES FuelType(FuelType_Id),
    Octane_Number NVARCHAR(5),
    Manufacturer NVARCHAR(100)
);

-- 3. Единица измерения
CREATE TABLE UnitOfMeasure (
    Unit_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(50) NOT NULL
);

-- 4. Категория
CREATE TABLE Category (
    Category_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(100) NOT NULL
);

-- 5. АЗС
CREATE TABLE GasStation (
    GasStation_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(100) NOT NULL,
    Address NVARCHAR(200),
    Phone NVARCHAR(25)
);

-- 6. Цистерна (1:1 к Топливо)
CREATE TABLE Cistern (
    Cistern_Id INT PRIMARY KEY IDENTITY,
    Total_Volume_Liters INT NOT NULL,
    Current_Level_Liters INT NOT NULL,
    Fuel_Id INT UNIQUE REFERENCES Fuel(Fuel_Id)
);

-- 7. Товар
-- ⚠️ ВАЖНО: Убираем UNIQUE с GasStation_Id, иначе 1 АЗС = 1 товар!
CREATE TABLE Product (
    Product_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(100) NOT NULL,
    Unit_Id INT REFERENCES UnitOfMeasure(Unit_Id),
    Category_Id INT REFERENCES Category(Category_Id),
    GasStation_Id INT REFERENCES GasStation(GasStation_Id), -- БЕЗ UNIQUE для M:1
    Stock_Quantity INT NOT NULL DEFAULT 0
);

-- 8. Роль
CREATE TABLE [Role] (
    Role_Id INT PRIMARY KEY IDENTITY,
    Name_Role NVARCHAR(50) NOT NULL  -- Исправлено: было Name
);

-- 9. Касса
CREATE TABLE CashRegister (
    CashRegister_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(50) NOT NULL,
    GasStation_Id INT REFERENCES GasStation(GasStation_Id)
);

-- 10. Работник
CREATE TABLE Worker (
    Worker_Id INT PRIMARY KEY IDENTITY,
    Full_Name NVARCHAR(200) NOT NULL,
    Phone_Number NVARCHAR(25),
    Passport_Number NVARCHAR(10),
    Role_Id INT REFERENCES [Role](Role_Id),
    CashRegister_Id INT REFERENCES CashRegister(CashRegister_Id),
    Hire_Date DATE NOT NULL,
    Login NVARCHAR(50) NOT NULL UNIQUE,
    Password NVARCHAR(100) NOT NULL
);

-- 11. Колонка
CREATE TABLE FuelPump (
    Pump_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(50) NOT NULL,
    Status NVARCHAR(20) DEFAULT 'Active',
    Cistern_Id INT REFERENCES Cistern(Cistern_Id),
    GasStation_Id INT REFERENCES GasStation(GasStation_Id)
);

-- 12. Топливный кран
CREATE TABLE FuelTap (
    Tap_Id INT PRIMARY KEY IDENTITY,
    Name NVARCHAR(20) NOT NULL,
    Max_Flow_Rate DECIMAL(5,2),
    Pump_Id INT REFERENCES FuelPump(Pump_Id)
);

-- 13. Чек
CREATE TABLE Receipt (
    Receipt_Id INT PRIMARY KEY IDENTITY,
    Worker_Id INT REFERENCES Worker(Worker_Id),
    Receipt_Date DATETIME NOT NULL DEFAULT GETDATE(),
    Total_Amount DECIMAL(10,2),
    Payment_Type NVARCHAR(20)
);

-- 14. ЧекТопливо
CREATE TABLE ReceiptFuel (
    ReceiptFuel_Id INT PRIMARY KEY IDENTITY,
    Receipt_Id INT REFERENCES Receipt(Receipt_Id),
    Fuel_Id INT REFERENCES Fuel(Fuel_Id),
    Quantity_Liters DECIMAL(10,2) NOT NULL,
    Price_Per_Liter DECIMAL(10,2) NOT NULL
);

-- 15. ЧекТовар
CREATE TABLE ReceiptProduct (
    ReceiptProduct_Id INT PRIMARY KEY IDENTITY,
    Receipt_Id INT REFERENCES Receipt(Receipt_Id),
    Product_Id INT REFERENCES Product(Product_Id),
    Quantity INT NOT NULL,
    Price_Per_Unit DECIMAL(10,2) NOT NULL
);

-- 16. Поставка
CREATE TABLE Delivery (
    Delivery_Id INT PRIMARY KEY IDENTITY,
    Worker_Id INT REFERENCES Worker(Worker_Id),
    Delivery_Date DATETIME NOT NULL DEFAULT GETDATE(),
    Status NVARCHAR(20) DEFAULT 'Pending'
);

-- 17. ПоставкаТовар
CREATE TABLE DeliveryProduct (
    DeliveryProduct_Id INT PRIMARY KEY IDENTITY,
    Delivery_Id INT REFERENCES Delivery(Delivery_Id),
    Product_Id INT REFERENCES Product(Product_Id),
    Quantity INT NOT NULL,
    Price_Per_Unit DECIMAL(10,2) NOT NULL
);
GO
